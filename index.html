Mobile SDK Client Implementation (Kotlin)

import android.os.Bundle
import androidx.appcompat.app.AppCompatActivity
import com.google.android.gms.ads.*
import com.google.android.gms.ads.interstitial.InterstitialAd
import com.google.android.gms.ads.interstitial.InterstitialAdLoadCallback
import com.google.android.gms.ads.rewarded.RewardedAd
import com.google.android.gms.ads.rewarded.RewardedAdLoadCallback

class AdManagerActivity : AppCompatActivity() {
    private var orchestratorInterstitial: InterstitialAd? = null
    private var harigovInterstitial: InterstitialAd? = null
    private var rewardedAd: RewardedAd? = null

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_ads)

        MobileAds.initialize(this) {}
        val request = AdRequest.Builder().build()

        // 1. Firebase (Banner)
        val firebaseBanner = AdView(this).apply {
            adUnitId = "ca-app-pub-5005820095625003/5556401390"
            setAdSize(AdSize.BANNER)
        }
        firebaseBanner.loadAd(request)

        // 2. vane-enterprise.github.io (Banner)
        val githubBanner = AdView(this).apply {
            adUnitId = "ca-app-pub-5005820095625003/1872916061"
            setAdSize(AdSize.BANNER)
        }
        githubBanner.loadAd(request)

        // 3. Vane-guard-sovereign-orchestrator (Interstitial)
        InterstitialAd.load(this, "ca-app-pub-5005820095625003/4307507716", request,
            object : InterstitialAdLoadCallback() {
                override fun onAdLoaded(ad: InterstitialAd) { orchestratorInterstitial = ad }
            })

        // 4. harigov63 (Interstitial)
        InterstitialAd.load(this, "ca-app-pub-5005820095625003/9779551583", request,
            object : InterstitialAdLoadCallback() {
                override fun onAdLoaded(ad: InterstitialAd) { harigovInterstitial = ad }
            })

        // 5. HARIG5E8D4 (Rewarded)
        RewardedAd.load(this, "ca-app-pub-5005820095625003/9832699505", request,
            object : RewardedAdLoadCallback() {
                override fun onAdLoaded(ad: RewardedAd) { rewardedAd = ad }
            })
    }
}
